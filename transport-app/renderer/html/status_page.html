💻 Complete ESP32 Arduino C++ Firmware Sketch

/*
 * JRR Transport - ESP32 Fleet Telemetry Firmware
 * Hardware: ESP32 + NEO-6M/8M GPS + DHT22 (Reefer Temp) + Reed Switch (Cargo Door)
 * Wiring:
 *   GPS: NEO-6M TX -> ESP32 GPIO16 (RX2), NEO-6M RX -> ESP32 GPIO17 (TX2)
 *   Cargo Temp: DHT22 Data -> GPIO4
 *   Cargo Door: Magnetic Reed Switch -> GPIO5 (INPUT_PULLUP)
 */

#include <WiFi.h>
#include <HTTPClient.h>
#include <TinyGPSPlus.h>
#include <DHT.h>

const char* ssid = "YOUR_WIFI_OR_HOTSPOT";
const char* password = "YOUR_WIFI_PASSWORD";
const char* serverUrl = "https://your-server.com/api/tracking/iot/telemetry";

const char* DEVICE_ID = "ESP32-FLT-TRK01";
const char* VEHICLE_PLATE = "TRK-8821";

#define DHTPIN 4
#define DHTTYPE DHT22
#define DOOR_PIN 5
#define RXD2 16
#define TXD2 17

TinyGPSPlus gps;
HardwareSerial gpsSerial(2);
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  gpsSerial.begin(9600, SERIAL_8N1, RXD2, TXD2);
  pinMode(DOOR_PIN, INPUT_PULLUP);
  dht.begin();

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");
}

void loop() {
  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }

  static unsigned long lastSend = 0;
  if (millis() - lastSend > 10000) { // Broadcast every 10 seconds
    lastSend = millis();

    float lat = gps.location.isValid() ? gps.location.lat() : 14.5468;
    float lng = gps.location.isValid() ? gps.location.lng() : 121.2293;
    float speed = gps.speed.isValid() ? gps.speed.kmph() : 0.0;
    int sats = gps.satellites.isValid() ? gps.satellites.value() : 8;

    float temp = dht.readTemperature();
    float humidity = dht.readHumidity();
    bool doorClosed = digitalRead(DOOR_PIN) == LOW; // LOW when magnet near

    if (WiFi.status() == WL_CONNECTED) {
      HTTPClient http;
      http.begin(serverUrl);
      http.addHeader("Content-Type", "application/json");

      String payload = "{";
      payload += "\"device_id\":\"" + String(DEVICE_ID) + "\",";
      payload += "\"vehicle_plate\":\"" + String(VEHICLE_PLATE) + "\",";
      payload += "\"latitude\":" + String(lat, 6) + ",";
      payload += "\"longitude\":" + String(lng, 6) + ",";
      payload += "\"speed\":" + String(speed, 1) + ",";
      payload += "\"satellites\":" + String(sats) + ",";
      payload += "\"cargo_temp\":" + String(temp, 1) + ",";
      payload += "\"cargo_humidity\":" + String(humidity, 1) + ",";
      payload += "\"cargo_door_closed\":" + String(doorClosed ? "true" : "false") + ",";
      payload += "\"battery_voltage\":12.7";
      payload += "}";

      int code = http.POST(payload);
      Serial.printf("Telemetry status: %d\n", code);
      http.end();
    }
  }
}
